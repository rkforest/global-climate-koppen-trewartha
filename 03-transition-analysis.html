Objective: Analyze changes in Köppen–Trewartha climate classifications among the REF, CSN, and RCI climate intervals. This notebook quantifies how much global land area changes climate group or type, identifies the most important transitions, summarizes net gains and losses by classification, and saves a unified transition dataset for geographic mapping in the final stage of the workflow.

Transition Comparisons

Three pairwise comparisons are used to separate historical, recent, and long-term classification change:

  • REF → CSN — change from the 1961–1990 WMO Climatological Reference Period to the 1991–2020 WMO Climatological Standard Normals.
  • CSN → RCI — change from the 1991–2020 standard normal to the 2021–2025 Recent Climate Interval.
  • REF → RCI — the full contrast between the reference period and the recent interval.

Transition calculations use land-area weighting based on the ERA5 grid-cell area multiplied by the fractional land-sea mask.

# creates the three grid-cell transition datasets
fn_build_transition <- function(
  classification_df,
  comparison_row
) {
  from_id <- comparison_row$from_period_id[[1]]
  to_id <- comparison_row$to_period_id[[1]]

  from_df <- classification_df |>
    filter(period_id == from_id) |>
    transmute(
      lon,
      lat,
      area,
      lsm,
      from_grp = as.character(grp),
      from_ktc = as.character(ktc)
    )

  to_df <- classification_df |>
    filter(period_id == to_id) |>
    transmute(
      lon,
      lat,
      to_area = area,
      to_lsm = lsm,
      to_grp = as.character(grp),
      to_ktc = as.character(ktc)
    )

  transition_df <- from_df |>
    inner_join(
      to_df,
      by = c("lon", "lat")
    )

  if (
    any(
      abs(transition_df$area - transition_df$to_area) > 1e-8,
      na.rm = TRUE
    ) ||
      any(
        abs(transition_df$lsm - transition_df$to_lsm) > 1e-8,
        na.rm = TRUE
      )
  ) {
    stop(
      "Grid-cell area or land-sea mask differs between ",
      comparison_row$comparison_label[[1]],
      " classification datasets."
    )
  }

  transition_df |>
    transmute(
      comparison_id = comparison_row$comparison_id[[1]],
      comparison_label = comparison_row$comparison_label[[1]],
      from_period_id = from_id,
      to_period_id = to_id,
      lon,
      lat,
      area,
      lsm,
      land_area = area * lsm,
      from_grp,
      to_grp,
      from_ktc,
      to_ktc,
      group_changed = from_grp != to_grp,
      type_changed = from_ktc != to_ktc
    ) |>
    filter(
      lsm > 0,
      is.finite(land_area),
      land_area > 0,
      !is.na(from_grp),
      !is.na(to_grp),
      !is.na(from_ktc),
      !is.na(to_ktc)
    )
}

# overall classification change summary
fn_change_overview <- function(transitions_df) {
  transitions_df |>
    group_by(
      comparison_id,
      comparison_label
    ) |>
    summarise(
      total_land_area = sum(land_area, na.rm = TRUE),
      group_changed_area = sum(
        land_area[group_changed],
        na.rm = TRUE
      ),
      type_changed_area = sum(
        land_area[type_changed],
        na.rm = TRUE
      ),
      .groups = "drop"
    ) |>
    mutate(
      group_changed_pct = group_changed_area / total_land_area,
      type_changed_pct = type_changed_area / total_land_area,
      comparison_label = factor(
        comparison_label,
        levels = comparison_levels
      )
    ) |>
    arrange(comparison_label)
}

# group/type transition summaries
fn_transition_summary <- function(
  transitions_df,
  from_variable,
  to_variable
) {
  transitions_df |>
    transmute(
      comparison_id,
      comparison_label,
      from_class = .data[[from_variable]],
      to_class = .data[[to_variable]],
      land_area
    ) |>
    group_by(
      comparison_id,
      comparison_label,
      from_class,
      to_class
    ) |>
    summarise(
      mkm = sum(land_area, na.rm = TRUE),
      .groups = "drop"
    ) |>
    group_by(
      comparison_id,
      comparison_label
    ) |>
    mutate(
      pct = mkm / sum(mkm)
    ) |>
    ungroup() |>
    mutate(
      changed = from_class != to_class,
      comparison_label = factor(
        comparison_label,
        levels = comparison_levels
      )
    ) |>
    arrange(
      comparison_label,
      desc(mkm)
    )
}

# group/type gains, losses, and net change
fn_net_change <- function(
  transition_summary,
  class_levels
) {
  from_summary <- transition_summary |>
    group_by(
      comparison_id,
      comparison_label,
      class = from_class
    ) |>
    summarise(
      from_mkm = sum(mkm),
      .groups = "drop"
    )

  to_summary <- transition_summary |>
    group_by(
      comparison_id,
      comparison_label,
      class = to_class
    ) |>
    summarise(
      to_mkm = sum(mkm),
      .groups = "drop"
    )

  gains <- transition_summary |>
    filter(changed) |>
    group_by(
      comparison_id,
      comparison_label,
      class = to_class
    ) |>
    summarise(
      gain_mkm = sum(mkm),
      .groups = "drop"
    )

  losses <- transition_summary |>
    filter(changed) |>
    group_by(
      comparison_id,
      comparison_label,
      class = from_class
    ) |>
    summarise(
      loss_mkm = sum(mkm),
      .groups = "drop"
    )

  full_join(
    from_summary,
    to_summary,
    by = c(
      "comparison_id",
      "comparison_label",
      "class"
    )
  ) |>
    full_join(
      gains,
      by = c(
        "comparison_id",
        "comparison_label",
        "class"
      )
    ) |>
    full_join(
      losses,
      by = c(
        "comparison_id",
        "comparison_label",
        "class"
      )
    ) |>
    complete(
      nesting(
        comparison_id,
        comparison_label
      ),
      class = class_levels
    ) |>
    mutate(
      across(
        c(
          from_mkm,
          to_mkm,
          gain_mkm,
          loss_mkm
        ),
        ~ replace_na(.x, 0)
      ),
      total_land_area = sum(from_mkm, na.rm = TRUE),
      from_pct = from_mkm / total_land_area,
      to_pct = to_mkm / total_land_area,
      gain_pct = gain_mkm / total_land_area,
      loss_pct = loss_mkm / total_land_area,
      net_pct = (to_mkm - from_mkm) / total_land_area,
      comparison_label = factor(
        comparison_label,
        levels = comparison_levels
      ),
      class = factor(
        class,
        levels = class_levels
      ),
      .by = c(
        comparison_id,
        comparison_label
      )
    ) |>
    arrange(
      comparison_label,
      class
    )
}

# build a transition matrices
fn_build_transition_matrix_gt <- function(
  transition_summary,
  comparison,
  class_levels,
  title,
  hide_diagonal = TRUE
) {
  matrix_df <- transition_summary |>
    filter(
      comparison_label ==
        comparison
    ) |>
    select(
      from_class,
      to_class,
      pct
    ) |>
    complete(
      from_class = class_levels,
      to_class = class_levels,
      fill = list(
        pct = 0
      )
    ) |>
    mutate(
      from_class = factor(
        from_class,
        levels = class_levels
      ),
      to_class = factor(
        to_class,
        levels = class_levels
      ),
      pct = if_else(
        hide_diagonal &
          as.character(from_class) ==
            as.character(to_class),
        NA_real_,
        pct
      )
    ) |>
    arrange(
      from_class,
      to_class
    ) |>
    pivot_wider(
      names_from = to_class,
      values_from = pct
    )

  matrix_df |>
    gt(
      rowname_col = "from_class"
    ) |>
    tab_header(
      title = title,
      subtitle = paste(
        comparison,
        "— percent of global land area"
      )
    ) |>
    tab_stubhead(
      label = "From"
    ) |>
    tab_spanner(
      label = "To",
      columns = all_of(
        class_levels
      )
    ) |>
    fmt_percent(
      columns = all_of(
        class_levels
      ),
      decimals = 2
    ) |>
    sub_missing(
      columns = all_of(
        class_levels
      ),
      missing_text = "—"
    )
}

# Largest Climate Type Transitions
fn_largest_transitions_gt <- function(
  transition_summary,
  title,
  n = 12
) {
  transition_summary |>
    filter(changed) |>
    group_by(
      comparison_id,
      comparison_label
    ) |>
    slice_max(
      order_by = mkm,
      n = n,
      with_ties = FALSE
    ) |>
    ungroup() |>
    mutate(
      transition = paste(
        from_class,
        "→",
        to_class
      )
    ) |>
    select(
      comparison_label,
      transition,
      mkm,
      pct
    ) |>
    arrange(
      comparison_label,
      desc(mkm)
    ) |>
    gt(groupname_col = "comparison_label") |>
    tab_header(
      title = title,
      subtitle = "Largest changed classifications by land area"
    ) |>
    cols_label(
      transition = "Transition",
      mkm = "Area (Mkm²)",
      pct = "Land Area %"
    ) |>
    fmt_number(
      columns = mkm,
      decimals = 2
    ) |>
    fmt_percent(
      columns = pct,
      decimals = 2
    )
}

fn_build_net_change_plot <- function(
  net_change_df,
  title,
  x_label
) {
  plot_df <- net_change_df |>
    left_join(
      transition_comparisons |>
        select(
          comparison_id,
          from_period_id,
          to_period_id
        ),
      by = "comparison_id"
    ) |>
    left_join(
      climate_periods |>
        select(
          period_id,
          from_abbr = period_abbr
        ),
      by = c(
        "from_period_id" = "period_id"
      )
    ) |>
    left_join(
      climate_periods |>
        select(
          period_id,
          to_abbr = period_abbr
        ),
      by = c(
        "to_period_id" = "period_id"
      )
    ) |>
    mutate(
      bar_period = if_else(
        net_pct >= 0,
        to_abbr,
        from_abbr
      ),
      bar_period = factor(
        bar_period,
        levels = climate_periods$period_abbr
      )
    )

  ggplot(
    plot_df,
    aes(
      x = class,
      y = net_pct,
      fill = bar_period
    )
  ) +
    geom_col() +
    geom_hline(
      yintercept = 0,
      linewidth = 0.4
    ) +
    facet_wrap(
      vars(comparison_label),
      ncol = 1
    ) +
    scale_y_continuous(
      labels = scales::percent_format(
        accuracy = 0.1
      )
    ) +
    scale_fill_manual(
      values = period_palette,
      breaks = climate_periods$period_abbr,
      drop = FALSE
    ) +
    labs(
      title = title,
      x = x_label,
      y = "Global Land Area %",
      fill = "Climate Interval"
    ) +
    theme_minimal(
      base_size = 12
    ) +
    theme(
      legend.position = "bottom",
      panel.grid.major.x = element_blank()
    )
}

**Objective:** Analyze changes in Köppen–Trewartha climate classifications among the REF, CSN, and RCI climate intervals. This notebook quantifies how much global land area changes climate group or type, identifies the most important transitions, summarizes net gains and losses by classification, and saves a unified transition dataset for geographic mapping in the final stage of the workflow.

## Transition Comparisons

Three pairwise comparisons are used to separate historical, recent, and long-term classification change:

- **REF → CSN** — change from the 1961–1990 WMO Climatological Reference Period to the 1991–2020 WMO Climatological Standard Normals.
- **CSN → RCI** — change from the 1991–2020 standard normal to the 2021–2025 Recent Climate Interval.
- **REF → RCI** — the full contrast between the reference period and the recent interval.

Transition calculations use land-area weighting based on the ERA5 grid-cell area multiplied by the fractional land-sea mask.

# creates the three grid-cell transition datasets
fn_build_transition <- function(
  classification_df,
  comparison_row
) {
  from_id <- comparison_row$from_period_id[[1]]
  to_id <- comparison_row$to_period_id[[1]]

  from_df <- classification_df |>
    filter(period_id == from_id) |>
    transmute(
      lon,
      lat,
      area,
      lsm,
      from_grp = as.character(grp),
      from_ktc = as.character(ktc)
    )

  to_df <- classification_df |>
    filter(period_id == to_id) |>
    transmute(
      lon,
      lat,
      to_area = area,
      to_lsm = lsm,
      to_grp = as.character(grp),
      to_ktc = as.character(ktc)
    )

  transition_df <- from_df |>
    inner_join(
      to_df,
      by = c("lon", "lat")
    )

  if (
    any(
      abs(transition_df$area - transition_df$to_area) > 1e-8,
      na.rm = TRUE
    ) ||
      any(
        abs(transition_df$lsm - transition_df$to_lsm) > 1e-8,
        na.rm = TRUE
      )
  ) {
    stop(
      "Grid-cell area or land-sea mask differs between ",
      comparison_row$comparison_label[[1]],
      " classification datasets."
    )
  }

  transition_df |>
    transmute(
      comparison_id = comparison_row$comparison_id[[1]],
      comparison_label = comparison_row$comparison_label[[1]],
      from_period_id = from_id,
      to_period_id = to_id,
      lon,
      lat,
      area,
      lsm,
      land_area = area * lsm,
      from_grp,
      to_grp,
      from_ktc,
      to_ktc,
      group_changed = from_grp != to_grp,
      type_changed = from_ktc != to_ktc
    ) |>
    filter(
      lsm > 0,
      is.finite(land_area),
      land_area > 0,
      !is.na(from_grp),
      !is.na(to_grp),
      !is.na(from_ktc),
      !is.na(to_ktc)
    )
}

# overall classification change summary
fn_change_overview <- function(transitions_df) {
  transitions_df |>
    group_by(
      comparison_id,
      comparison_label
    ) |>
    summarise(
      total_land_area = sum(land_area, na.rm = TRUE),
      group_changed_area = sum(
        land_area[group_changed],
        na.rm = TRUE
      ),
      type_changed_area = sum(
        land_area[type_changed],
        na.rm = TRUE
      ),
      .groups = "drop"
    ) |>
    mutate(
      group_changed_pct = group_changed_area / total_land_area,
      type_changed_pct = type_changed_area / total_land_area,
      comparison_label = factor(
        comparison_label,
        levels = comparison_levels
      )
    ) |>
    arrange(comparison_label)
}

# group/type transition summaries
fn_transition_summary <- function(
  transitions_df,
  from_variable,
  to_variable
) {
  transitions_df |>
    transmute(
      comparison_id,
      comparison_label,
      from_class = .data[[from_variable]],
      to_class = .data[[to_variable]],
      land_area
    ) |>
    group_by(
      comparison_id,
      comparison_label,
      from_class,
      to_class
    ) |>
    summarise(
      mkm = sum(land_area, na.rm = TRUE),
      .groups = "drop"
    ) |>
    group_by(
      comparison_id,
      comparison_label
    ) |>
    mutate(
      pct = mkm / sum(mkm)
    ) |>
    ungroup() |>
    mutate(
      changed = from_class != to_class,
      comparison_label = factor(
        comparison_label,
        levels = comparison_levels
      )
    ) |>
    arrange(
      comparison_label,
      desc(mkm)
    )
}

# group/type gains, losses, and net change
fn_net_change <- function(
  transition_summary,
  class_levels
) {
  from_summary <- transition_summary |>
    group_by(
      comparison_id,
      comparison_label,
      class = from_class
    ) |>
    summarise(
      from_mkm = sum(mkm),
      .groups = "drop"
    )

  to_summary <- transition_summary |>
    group_by(
      comparison_id,
      comparison_label,
      class = to_class
    ) |>
    summarise(
      to_mkm = sum(mkm),
      .groups = "drop"
    )

  gains <- transition_summary |>
    filter(changed) |>
    group_by(
      comparison_id,
      comparison_label,
      class = to_class
    ) |>
    summarise(
      gain_mkm = sum(mkm),
      .groups = "drop"
    )

  losses <- transition_summary |>
    filter(changed) |>
    group_by(
      comparison_id,
      comparison_label,
      class = from_class
    ) |>
    summarise(
      loss_mkm = sum(mkm),
      .groups = "drop"
    )

  full_join(
    from_summary,
    to_summary,
    by = c(
      "comparison_id",
      "comparison_label",
      "class"
    )
  ) |>
    full_join(
      gains,
      by = c(
        "comparison_id",
        "comparison_label",
        "class"
      )
    ) |>
    full_join(
      losses,
      by = c(
        "comparison_id",
        "comparison_label",
        "class"
      )
    ) |>
    complete(
      nesting(
        comparison_id,
        comparison_label
      ),
      class = class_levels
    ) |>
    mutate(
      across(
        c(
          from_mkm,
          to_mkm,
          gain_mkm,
          loss_mkm
        ),
        ~ replace_na(.x, 0)
      ),
      total_land_area = sum(from_mkm, na.rm = TRUE),
      from_pct = from_mkm / total_land_area,
      to_pct = to_mkm / total_land_area,
      gain_pct = gain_mkm / total_land_area,
      loss_pct = loss_mkm / total_land_area,
      net_pct = (to_mkm - from_mkm) / total_land_area,
      comparison_label = factor(
        comparison_label,
        levels = comparison_levels
      ),
      class = factor(
        class,
        levels = class_levels
      ),
      .by = c(
        comparison_id,
        comparison_label
      )
    ) |>
    arrange(
      comparison_label,
      class
    )
}

# build a transition matrices
fn_build_transition_matrix_gt <- function(
  transition_summary,
  comparison,
  class_levels,
  title,
  hide_diagonal = TRUE
) {
  matrix_df <- transition_summary |>
    filter(
      comparison_label ==
        comparison
    ) |>
    select(
      from_class,
      to_class,
      pct
    ) |>
    complete(
      from_class = class_levels,
      to_class = class_levels,
      fill = list(
        pct = 0
      )
    ) |>
    mutate(
      from_class = factor(
        from_class,
        levels = class_levels
      ),
      to_class = factor(
        to_class,
        levels = class_levels
      ),
      pct = if_else(
        hide_diagonal &
          as.character(from_class) ==
            as.character(to_class),
        NA_real_,
        pct
      )
    ) |>
    arrange(
      from_class,
      to_class
    ) |>
    pivot_wider(
      names_from = to_class,
      values_from = pct
    )

  matrix_df |>
    gt(
      rowname_col = "from_class"
    ) |>
    tab_header(
      title = title,
      subtitle = paste(
        comparison,
        "— percent of global land area"
      )
    ) |>
    tab_stubhead(
      label = "From"
    ) |>
    tab_spanner(
      label = "To",
      columns = all_of(
        class_levels
      )
    ) |>
    fmt_percent(
      columns = all_of(
        class_levels
      ),
      decimals = 2
    ) |>
    sub_missing(
      columns = all_of(
        class_levels
      ),
      missing_text = "—"
    )
}

# Largest Climate Type Transitions
fn_largest_transitions_gt <- function(
  transition_summary,
  title,
  n = 12
) {
  transition_summary |>
    filter(changed) |>
    group_by(
      comparison_id,
      comparison_label
    ) |>
    slice_max(
      order_by = mkm,
      n = n,
      with_ties = FALSE
    ) |>
    ungroup() |>
    mutate(
      transition = paste(
        from_class,
        "→",
        to_class
      )
    ) |>
    select(
      comparison_label,
      transition,
      mkm,
      pct
    ) |>
    arrange(
      comparison_label,
      desc(mkm)
    ) |>
    gt(groupname_col = "comparison_label") |>
    tab_header(
      title = title,
      subtitle = "Largest changed classifications by land area"
    ) |>
    cols_label(
      transition = "Transition",
      mkm = "Area (Mkm²)",
      pct = "Land Area %"
    ) |>
    fmt_number(
      columns = mkm,
      decimals = 2
    ) |>
    fmt_percent(
      columns = pct,
      decimals = 2
    )
}

if (run_transition_analysis) {
